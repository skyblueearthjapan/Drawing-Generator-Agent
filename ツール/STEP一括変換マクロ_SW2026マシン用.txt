SolidWorks用 SLDPRT→STEP一括変換マクロ(会社のSW2026マシンで実行する版)
====================================================================

背景: 荏原トライ調整用の3Dモデル(SLDPRT)はSolidWorks 2026で保存されており、
開発機のSW2023では開けない(SolidWorksの後方互換は無い)。
→ SW2026のあるマシンでこのマクロを実行し、「教師STEP」フォルダを作って
   開発機の 部品図作成agent\荏原トライ調整用\ 配下にコピーしてください。

変換元(SRC_ROOT): \\lineworks-sv\Data\技術部\機械設計\01AMD-D\社内工程\25154：荏原エリオット袖ケ浦工場／配管仮組位置決め装置(LINEMAN2000)\３Ｄ\取り合い確認用3D
出力先(OUT_ROOT): 同じ「３Ｄ」フォルダ直下に「教師STEP」を新規作成(元データには一切書き込みません)

使い方:
1. SW2026マシンで SolidWorks を起動(部品は開かなくてOK)
   ※そのマシンから \\lineworks-sv\Data の共有が見えることを確認してください
2. メニュー「ツール > マクロ > 新規作成」→ 適当な名前で保存(VBAエディタが開く)
3. 開いたエディタの中身を全部消して、下のコードを貼り付け
   ※パスに全角の「：」「／」「３Ｄ」が含まれます。書き換えず、このままコピーしてください
4. F5(実行)を押す
5. 「取り合い確認用3D」内の全SLDPRTが「教師STEP」にサブフォルダ構成ごと変換されます
   (点数により数分〜十数分。完了するとメッセージと変換ログが出ます)
6. できた「教師STEP」フォルダごと、開発機の
   C:\Users\imaizumi.LINEWORKS-NET\Documents\部品図作成agent\荏原トライ調整用\
   にコピーしてください

-----------------------------------------(ここから下をコピー)

Option Explicit

Const SRC_ROOT As String = "\\lineworks-sv\Data\技術部\機械設計\01AMD-D\社内工程\25154：荏原エリオット袖ケ浦工場／配管仮組位置決め装置(LINEMAN2000)\３Ｄ\取り合い確認用3D"
Const OUT_ROOT As String = "\\lineworks-sv\Data\技術部\機械設計\01AMD-D\社内工程\25154：荏原エリオット袖ケ浦工場／配管仮組位置決め装置(LINEMAN2000)\３Ｄ\教師STEP"

Dim swApp As Object
Dim fso As Object
Dim nOK As Long, nNG As Long, logText As String

Sub main()
    Set swApp = Application.SldWorks
    Set fso = CreateObject("Scripting.FileSystemObject")
    nOK = 0: nNG = 0: logText = ""

    If Not fso.FolderExists(SRC_ROOT) Then
        MsgBox "変換元フォルダが見えません: " & SRC_ROOT & vbCrLf & _
               "共有 \\lineworks-sv\Data への接続を確認してください。"
        Exit Sub
    End If

    If Not fso.FolderExists(OUT_ROOT) Then fso.CreateFolder OUT_ROOT
    ProcessFolder fso.GetFolder(SRC_ROOT), OUT_ROOT

    Dim logPath As String
    logPath = OUT_ROOT & "\変換ログ.txt"
    Dim ts As Object
    Set ts = fso.CreateTextFile(logPath, True, True)
    ts.Write "成功: " & nOK & " / 失敗: " & nNG & vbCrLf & vbCrLf & logText
    ts.Close

    MsgBox "変換完了: 成功 " & nOK & " 件 / 失敗 " & nNG & " 件" & vbCrLf & _
           "ログ: " & logPath
End Sub

Sub ProcessFolder(fld As Object, outDir As String)
    If Not fso.FolderExists(outDir) Then fso.CreateFolder outDir

    Dim f As Object
    For Each f In fld.Files
        If LCase(fso.GetExtensionName(f.Path)) = "sldprt" Then
            ConvertOne f.Path, outDir & "\" & fso.GetBaseName(f.Path) & ".STEP"
        End If
    Next

    Dim sub_ As Object
    For Each sub_ In fld.SubFolders
        ' 出力先(教師STEP)自身を再帰対象にしない(SRC=OUTの親が同じため念のため)
        If StrComp(sub_.Path, OUT_ROOT, vbTextCompare) <> 0 Then
            ProcessFolder sub_, outDir & "\" & sub_.Name
        End If
    Next
End Sub

Sub ConvertOne(srcPath As String, outPath As String)
    On Error Resume Next
    Dim errs As Long, warns As Long
    Dim doc As Object
    ' 1 = swDocPART, 1 = silent
    Set doc = swApp.OpenDoc6(srcPath, 1, 1, "", errs, warns)
    If doc Is Nothing Then
        nNG = nNG + 1
        logText = logText & "NG(開けない): " & srcPath & vbCrLf
        Exit Sub
    End If
    Dim ok As Boolean
    ok = doc.Extension.SaveAs(outPath, 0, 0, Nothing, errs, warns)
    swApp.CloseDoc doc.GetTitle
    If ok Then
        nOK = nOK + 1
    Else
        nNG = nNG + 1
        logText = logText & "NG(保存失敗 err=" & errs & "): " & srcPath & vbCrLf
    End If
End Sub
